SUB_ROTINA cadastrar_produto
DECLARE NOME, CATEGORIA LITERAL
DECLARE PRECO, QUANTIDADE, POSICAO NUMÉRICO
SE TOTAL_PRODUTOS = 100
ENTÃO
    ESCREVA "Limite de produtos atingido."
SENÃO
    ESCREVA "Nome do produto: "
    LEIA NOME
    POSICAO ← buscar_produto(NOME)
    SE POSICAO <> -1
    ENTÃO
        ESCREVA "Já existe um produto com esse nome."
    SENÃO
        CATEGORIA ← ler_categoria()
        REPITA
            ESCREVA "Preço do produto: "
            LEIA PRECO
            SE PRECO < 0
            ENTÃO ESCREVA "O preço não pode ser negativo."
        ATÉ PRECO >= 0
        REPITA
            ESCREVA "Quantidade em estoque: "
            LEIA QUANTIDADE
            SE QUANTIDADE < 0
            ENTÃO ESCREVA "A quantidade não pode ser negativa."
        ATÉ QUANTIDADE >= 0
        TOTAL_PRODUTOS ← TOTAL_PRODUTOS + 1
        NOMES[TOTAL_PRODUTOS] ← NOME
        CATEGORIAS[TOTAL_PRODUTOS] ← CATEGORIA
        PRECOS[TOTAL_PRODUTOS] ← PRECO
        QUANTIDADES[TOTAL_PRODUTOS] ← QUANTIDADE
        ESCREVA "Produto cadastrado com sucesso."
    FIM
FIM
FIM_SUB_ROTINA cadastrar_produto.

SUB_ROTINA editar_produto

DECLARE POSICAO, POSICAO_NOME, I NUMÉRICO
DECLARE NOVO_PRECO, NOVA_QUANTIDADE NUMÉRICO

LEIA NOME
POSICAO ← buscar_produto(NOME)
SE POSICAO = -1
ENTÃO
    ESCREVA "Produto não encontrado."
SENÃO
    ESCREVA "Novo nome do produto: "
    LEIA NOVO_NOME
    POSICAO_NOME ← buscar_produto(NOVO_NOME)
    SE POSICAO_NOME <> -1 E POSICAO_NOME <> POSICAO
    ENTÃO
        ESCREVA "Já existe outro produto com esse nome."
    SENÃO
        CATEGORIA ← ler_categoria()
        REPITA

            LEIA NOVO_PRECO
            SE NOVO_PRECO < 0
            ENTÃO ESCREVA "O preço não pode ser negativo."
        ATÉ NOVO_PRECO >= 0
        REPITA
            ESCREVA "Nova quantidade: "
            LEIA NOVA_QUANTIDADE
            SE NOVA_QUANTIDADE < 0
            ENTÃO ESCREVA "A quantidade não pode ser negativa."
        ATÉ NOVA_QUANTIDADE >= 0
        NOMES[POSICAO] ← NOVO_NOME
        CATEGORIAS[POSICAO] ← CATEGORIA
        PRECOS[POSICAO] ← NOVO_PRECO
        QUANTIDADES[POSICAO] ← NOVA_QUANTIDADE

    FIM
FIM

SUB_ROTINA remover_produto

        CATEGORIAS[I] ← CATEGORIAS[I + 1]
        PRECOS[I] ← PRECOS[I + 1]
        QUANTIDADES[I] ← QUANTIDADES[I + 1]
    FIM_PARA
    TOTAL_PRODUTOS ← TOTAL_PRODUTOS - 1
    ESCREVA "Produto removido com sucesso."
FIM
FIM_SUB_ROTINA remover_produto.
